# PyInstaller version resource for ChessHindsight.exe.
#
# Without this the exe carries no VERSIONINFO at all, so Explorer's Details tab
# and the "Description" column are blank and the file is identified only by its
# filename. FileDescription is the string Explorer and Task Manager show.
VSVersionInfo(
    ffi=FixedFileInfo(
        filevers=(0, 1, 0, 0),
        prodvers=(0, 1, 0, 0),
        mask=0x3F,
        flags=0x0,
        OS=0x40004,
        fileType=0x1,
        subtype=0x0,
        date=(0, 0),
    ),
    kids=[
        StringFileInfo(
            [
                StringTable(
                    # 0409 = US English, 04B0 = Unicode codepage.
                    "040904B0",
                    [
                        StringStruct("CompanyName", "Thomas Murray"),
                        StringStruct("FileDescription", "ChessHindsight"),
                        StringStruct("FileVersion", "0.1.0.0"),
                        StringStruct("InternalName", "ChessHindsight"),
                        StringStruct("LegalCopyright", "Copyright (c) Thomas Murray"),
                        StringStruct("OriginalFilename", "ChessHindsight.exe"),
                        StringStruct("ProductName", "ChessHindsight"),
                        StringStruct("ProductVersion", "0.1.0.0"),
                    ],
                )
            ]
        ),
        VarFileInfo([VarStruct("Translation", [0x409, 1200])]),
    ],
)
